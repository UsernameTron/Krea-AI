```javascript
import Anthropic from '@anthropic-ai/sdk';
import { classifyInput } from './input-classifier.js';
import { buildSystemPrompt, buildUserMessage } from './prompts/system-prompt-builder.js';
import { STRUCTURED_OUTPUTS_SCHEMA } from './structured-outputs-schema.js';
import { validateBrief } from './schema.js';

const DEFAULT_MODEL = "claude-opus-4-6";  // D-22
const MAX_TOKENS = 4096;

// Error codes per D-19
export class BriefGenerationError extends Error {
  constructor(code, message, details) {
    super(message);
    this.name = "BriefGenerationError";
    this.code = code;
    this.details = details;
  }
}

export async function generateBrief(rawInput, options = {}) {
  const model = options.model || DEFAULT_MODEL;
  const client = options.client || new Anthropic();  // Allow mock injection

  // Step 1: Classify input
  const classified = classifyInput(rawInput);

  // Step 2: Build prompts
  const systemPrompt = buildSystemPrompt(classified.archetype);
  const userMessage = buildUserMessage(classified);

  // Step 3: Primary path — Structured Outputs (D-01)
  try {
    const response = await client.messages.create({
      model,
      max_tokens: MAX_TOKENS,
      system: systemPrompt,
      messages: [{ role: "user", content: userMessage }],
      output_config: { format: STRUCTURED_OUTPUTS_SCHEMA }  // D-01: constrained decoding
    });

    // D-01: response arrives in content[0].text as JSON string
    const brief = JSON.parse(response.content[0].text);

    // Structural validation (always, even with Structured Outputs — D-02)
    const validation = validateBrief(brief);
    if (!validation.valid) {
      throw new BriefGenerationError(
        "SCHEMA_VALIDATION_FAILED",
        `Generated brief failed schema validation: ${validation.errors.join(", ")}`,
        { errors: validation.errors, brief }
      );
    }

    // Inject meta fields that the LLM may not set correctly
    brief.meta.generated_at = new Date().toISOString();
    brief.meta.persona_spec_version = "mirror_pete.v2";
    brief.meta.input_type = classified.input_type;
    brief.meta.input_raw = classified.input_raw;

    return { brief, usage: response.usage, model: response.model };

  } catch (err) {
    if (err instanceof BriefGenerationError) throw err;

    // Step 4: Fallback path — plain text + parse (D-03)
    return await fallbackGeneration(client, model, systemPrompt, userMessage, classified);
  }
}

async function fallbackGeneration(client, model, systemPrompt, userMessage, classified) {
  // D-18: 1 retry with correction hint, then throw
  for (let attempt = 0; attempt < 2; attempt++) {
    const messages = [{ role: "user", content: userMessage }];

    if (attempt > 0) {
      messages.push({
        role: "assistant",
        content: "I'll generate the Post Brief JSON now."
      });
      messages.push({
        role: "user",
        content: [{ type: "text", text: "Return ONLY a valid JSON object matching the Post Brief v1 schema. No markdown, no explanation, just the JSON." }]
      });
    }

    const response = await client.messages.create({
      model,
      max_tokens: MAX_TOKENS,
      system: systemPrompt,
      messages
    });

    try {
      const text = response.content[0].text;
      // Try to extract JSON from response (may be wrapped in markdown code block)
      const jsonMatch = text.match(/\{[\s\S]*\}/);
      if (!jsonMatch) throw new Error("No JSON object found in response");

      const brief = JSON.parse(jsonMatch[0]);
      const validation = validateBrief(brief);
      if (!validation.valid) {
        if (attempt === 0) continue;  // Retry once
        throw new BriefGenerationError(
          "FALLBACK_VALIDATION_FAILED",
          `Fallback generation failed validation after retry: ${validation.errors.join(", ")}`,
          { errors: validation.errors }
        );
      }

      brief.meta.generated_at = new Date().toISOString();
      brief.meta.persona_spec_version = "mirror_pete.v2";
      brief.meta.input_type = classified.input_type;
      brief.meta.input_raw = classified.input_raw;

      return { brief, usage: response.usage, model: response.model, fallback: true };

    } catch (parseErr) {
      if (attempt === 0) continue;  // Retry once
      throw new BriefGenerationError(
        "PARSE_FAILED",
        `Could not parse JSON from LLM response after retry: ${parseErr.message}`,
        { rawResponse: response.content[0].text }
      );
    }
  }
}
```
