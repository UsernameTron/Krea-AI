    ```javascript
    /**
     * Golden test — REQ-X-052b overlay determinism.
     * Verifies byte-identical compositor output given identical inputs.
     *
     * Usage:
     *   node test/compositor-golden-v2.js              # Compare against committed golden
     *   REGENERATE_GOLDEN=1 node test/compositor-golden-v2.js  # Regenerate golden fixture
     *
     * External dependencies:
     *   - Chrome PNG: mirror-post/src/compositor/chrome-assets/variant-{x}.png (D-NEW-09)
     *   - Scene PNG: mirror-post/test/fixtures/scenes/golden-test-scene.png
     *     (synthetic fixture OR Phase 5a delivered fixture)
     */

    import { readFileSync, writeFileSync, existsSync, mkdirSync } from 'node:fs';
    import { join, dirname } from 'node:path';
    import { fileURLToPath } from 'node:url';
    import sharp from 'sharp';

    const __dirname = dirname(fileURLToPath(import.meta.url));
    const FIXTURES_DIR = join(__dirname, 'fixtures');
    const GOLDEN_DIR = join(FIXTURES_DIR, 'golden-outputs');
    const SCENES_DIR = join(FIXTURES_DIR, 'scenes');

    const GOLDEN_PATH = join(GOLDEN_DIR, 'brent-vellum-v2-golden.png');
    const SCENE_PATH = join(SCENES_DIR, 'golden-test-scene.png');
    const BRIEF_PATH = join(FIXTURES_DIR, 'expected-briefs', 'brent-vellum.json');

    // Resolve chrome asset path based on the brief's variant
    const CHROME_BASE = join(__dirname, '..', 'src', 'compositor', 'chrome-assets');

    async function ensureSyntheticScene() {
      // If no scene fixture exists, generate a 1920x1080 solid-color PNG.
      // This allows the golden test to run before Phase 5a delivers real scenes.
      if (!existsSync(SCENE_PATH)) {
        console.log('GENERATING SYNTHETIC SCENE FIXTURE (1920x1080 solid #2C3E50)');
        const buf = await sharp({
          create: { width: 1920, height: 1080, channels: 4, background: { r: 44, g: 62, b: 80, alpha: 1 } }
        }).png().toBuffer();
        mkdirSync(dirname(SCENE_PATH), { recursive: true });
        writeFileSync(SCENE_PATH, buf);
      }
    }

    async function main() {
      const regenerate = process.env.REGENERATE_GOLDEN === '1';

      // Load and patch the v2 brief to point at the golden test scene
      if (!existsSync(BRIEF_PATH)) {
        console.error(`MISSING FIXTURE: ${BRIEF_PATH}`);
        process.exit(1);
      }
      const brief = JSON.parse(readFileSync(BRIEF_PATH, 'utf-8'));

      // Override scene_png to use the golden test scene (absolute path)
      await ensureSyntheticScene();
      brief.image_output = { scene_png: SCENE_PATH };

      // Resolve variant for chrome check
      const variant = brief.image_seed?.scene_template ?? 'B';
      const chromePath = join(CHROME_BASE, `variant-${variant.toLowerCase()}.png`);
      if (!existsSync(chromePath)) {
        console.error(`MISSING FIXTURE: Chrome PNG not found at ${chromePath}`);
        console.error('Chrome PNGs are created by operator handoff (D-NEW-09). Run chrome authoring first.');
        process.exit(1);
      }

      // Import compositor (dynamic import to allow for ESM)
      const { compositePost } = await import('../src/compositor/composite.js');

      // Generate output
      console.log(`Compositing brent-vellum v2 (variant ${variant})...`);
      const output = await compositePost(brief);

      // Verify output dimensions
      const meta = await sharp(output).metadata();
      if (meta.width !== 1920 || meta.height !== 1080) {
        console.error(`OUTPUT DIMENSION MISMATCH: ${meta.width}x${meta.height} (expected 1920x1080)`);
        process.exit(1);
      }

      if (regenerate) {
        // Regenerate mode: overwrite golden fixture
        mkdirSync(GOLDEN_DIR, { recursive: true });
        writeFileSync(GOLDEN_PATH, output);
        console.log(`GOLDEN REGENERATED: ${GOLDEN_PATH} (${output.length} bytes)`);

        // Self-consistency check: run again and compare
        const output2 = await compositePost(brief);
        if (Buffer.compare(output, output2) !== 0) {
          console.error('DETERMINISM FAILURE: Two consecutive runs produced different output');
          process.exit(1);
        }
        console.log('SELF-CONSISTENCY: PASS (two runs identical)');
        process.exit(0);
      }

      // Comparison mode: compare against committed golden
      if (!existsSync(GOLDEN_PATH)) {
        console.error(`MISSING FIXTURE: Golden output not found at ${GOLDEN_PATH}`);
        console.error('Run with REGENERATE_GOLDEN=1 to generate the golden fixture first.');
        process.exit(1);
      }

      const golden = readFileSync(GOLDEN_PATH);
      const cmp = Buffer.compare(output, golden);
      if (cmp === 0) {
        console.log(`GOLDEN TEST: PASS (${output.length} bytes, byte-identical)`);

        // Self-consistency bonus check
        const output2 = await compositePost(brief);
        if (Buffer.compare(output, output2) !== 0) {
          console.error('DETERMINISM FAILURE: Two consecutive runs produced different output');
          process.exit(1);
        }
        console.log('SELF-CONSISTENCY: PASS');
        process.exit(0);
      } else {
        console.error('GOLDEN TEST: FAIL (output differs from golden fixture)');
        console.error(`  Output:  ${output.length} bytes`);
        console.error(`  Golden:  ${golden.length} bytes`);
        console.error('If this is expected (VARIANT_SLOTS or chrome changed), regenerate with:');
        console.error('  REGENERATE_GOLDEN=1 node test/compositor-golden-v2.js');
        process.exit(1);
      }
    }

    main().catch(err => {
      console.error('GOLDEN TEST ERROR:', err.message);
      process.exit(1);
    });
    ```

    **Key design choices:**
    - Synthetic scene fixture (solid color) unblocks the golden test before Phase 5a. When real scene fixtures arrive, the golden can be regenerated with `REGENERATE_GOLDEN=1`.
    - Brief's `image_output.scene_png` is overridden in-test to point at the known fixture path. This decouples the golden test from fixture path conventions in the brief JSON.
    - Self-consistency check (two runs produce identical output) validates determinism even when no golden exists.
    - Clear "MISSING FIXTURE" errors with actionable guidance for each dependency (scene PNG, chrome PNG, golden output).
    - Golden fixture committed to git at `test/fixtures/golden-outputs/brent-vellum-v2-golden.png`.

    **Execution note:** After the test file is written, the executor should:
    1. Generate the synthetic scene fixture (the test does this automatically)
    2. Run `REGENERATE_GOLDEN=1 node test/compositor-golden-v2.js` to generate the initial golden
    3. Commit the golden PNG fixture
    4. Run `node test/compositor-golden-v2.js` to confirm comparison passes

    This is blocked on chrome PNGs existing (D-NEW-09 operator handoff). If chrome PNGs are not yet committed, the test will exit with a clear error. The test file itself can be written and committed now; execution waits for chrome assets.
